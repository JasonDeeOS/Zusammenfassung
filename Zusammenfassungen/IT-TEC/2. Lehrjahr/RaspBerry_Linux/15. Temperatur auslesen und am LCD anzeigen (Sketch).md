```C
// Einbinden der benötigten Bibliotheken
#include <M5StickC.H>
#include <Wire.h>
#include "SHT30.h"
// Deklarieren und Initialisieren der globalen Variablen für Zeiten, Wert und Temperaturwert
unsigned long now = millis();
unsigned long lastMsg = 0;
int value = 0;
float tmp = 0.0;
// Erzeugen eines globalen Objektes der Klasse SHT30, um Temperaturwerte auslesen zu können
SHT30 sht30Sensor;
// Rumpf für einmal ausgeführte setup()-Funktion
void setup()
{
	// Initialisieren des M5Stick
	M5.begin();
	// Festlegen der Hintergrundfarbe, Textfarbe und Ausrichtung des LCD
	M5.Lcd.fillScreen(BLACK);
	M5.Lcd.setTextColor(WHITE);
	M5.Lcd.setRotation(3);
	//Initialisieren der Kommunikation mittels I2C-Bus
	Wire.begin();
}
//Rumpf für für die in Endlosschleife ausgeführte loop()-Funktion
//Verantwortlich für die Hauptlogik des Programms
void loop() 
{
	now = millis();
	//Abfrage, ob seit dem letzten Wert 5 Sekunden vergangen sind
	//Falls ja, Inhalt der Verzweigung ausführen
	if (now - lastMsg > 5000) 
	{
		lastMsg = now;
		++value;
		//Bereinigen des LCD nach jedem 5.Wert
		if (value % 5 == 0) 
		{
			M5.Lcd.fillScreen(BLACK);
			M5.Lcd.setCursor(0, 0);
		}
		//Temperatur auslesen
		tmp = sht30Sensor.readTemperature();
		//Ausgabe der Temperatur am seriellen Monitor
		Serial.print("Temperatur: ");
		Serial.print(tmp);
		Serial.println();
		Serial.println();
		//Ausgabe der Temperatur am LCD
		M5.Lcd.print("Temp: ");
		M5.Lcd.println(tmp);
		M5.Lcd.println();
	}
}
```
