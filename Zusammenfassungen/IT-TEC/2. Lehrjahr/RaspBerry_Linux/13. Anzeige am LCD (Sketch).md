``` C
#include <M5StickC.h> // Bibliothek für Nutzung des M5Sticks

int counter = 0; // globale Variable mit 0 initialisiert

void setup() // einmalig ausgeführt
{ 
	M5.begin(); // initialisieren des Sticks
	M5.Lcd.setTextSize(4); // Festlegen der Schriftgröße
	M5.Lcd.setTextColor(TFT_WHITE); // Festlegen der Textfarbe
	M5.Lcd.fillScreen(TFT_BLACK); // Festlegen der Hintergrundfarbe
}

void loop() // Hauptlogik in Endlosschleife
{
	M5.update(); // liest Veränderung am M5Stick
	
	if(M5.BtnA.wasPressed()) // prüft Betätigung Button A
	{
		counter++; // Counter +1
		updateCounterDisplay(counter); // Zählerstand am Display ausgeben
		Serial.print("Counter = "); // Zählerstand am seriellen Monitor ausgeben
		Serial.println(counter); // Zählerstand am seriellen Monitor ausgeben
	}
	if(M5.BtnB.wasPressed()) // prüft Betätigung Button B
	{
		counter--; // Counter -1
		updateCounterDisplay(counter); // Zählerstand am Display ausgeben
		Serial.print("Counter = "); // Zählerstand am seriellen Monitor ausgeben
		Serial.println(counter); // Zählerstand am seriellen Monitor ausgeben
	}
}

void updateCounterDisplay(int wert) // eigene Funktion zur Ausgabe des Zählerwerts mit Position 10, 10 und Font 2
{
	M5.Lcd.fillScreen(TFT_BLACK);
	M5.Lcd.setCursor(10,10,2);
	M5.Lcd.print(wert);
}
```
